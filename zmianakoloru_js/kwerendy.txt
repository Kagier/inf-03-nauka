1. SELECT imie, pensja FROM pracownicy WHERE stanowiska_id < 5;
2. SELECT AVG(pensja), nazwa FROM pracownicy p JOIN stanowiska s ON p.stanowiska_id = s.id GROUP BY s.nazwa;
3. SELECT imie, nazwisko, pensja FROM pracownicy WHERE pensja = (SELECT MAX(pensja) FROM pracownicy);
4. UPDATE pracownicy SET staz = staz + 1;
